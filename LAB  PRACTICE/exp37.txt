
#include <stdio.h>
#include <stdlib.h>

int main()
{
    int n, i, head, total = 0;
    int request[20];

    printf("Enter number of disk requests: ");
    scanf("%d", &n);

    printf("Enter disk requests:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &request[i]);
    }

    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("\nDisk head movement:\n");

    for(i = 0; i < n; i++)
    {
        printf("%d -> ", head);

        total += abs(request[i] - head);
        head = request[i];
    }

    printf("%d\n", head);

    printf("\nTotal head movement = %d\n", total);

    return 0;
}
