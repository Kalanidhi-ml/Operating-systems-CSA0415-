
#include <stdio.h>

int main()
{
    int index[20], n, i, block;

    printf("Enter number of file blocks: ");
    scanf("%d", &n);

    printf("Enter the block numbers:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &index[i]);
    }

    printf("\nIndex Block\n");
    printf("Index\tBlock Number\n");

    for(i = 0; i < n; i++)
    {
        printf("%d\t%d\n", i, index[i]);
    }

    printf("\nEnter index to access: ");
    scanf("%d", &block);

    if(block >= 0 && block < n)
    {
        printf("File block at index %d is %d\n",
               block, index[block]);
    }
    else
    {
        printf("Invalid index\n");
    }

    return 0;
}
