
#include <stdio.h>
#include <stdlib.h>

int main()
{
    int req[20], n, head, size;
    int i, j, temp, total = 0;

    printf("Enter number of requests: ");
    scanf("%d", &n);

    printf("Enter disk requests:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &req[i]);

    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("Enter disk size: ");
    scanf("%d", &size);

    // Sort requests
    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(req[j] > req[j + 1])
            {
                temp = req[j];
                req[j] = req[j + 1];
                req[j + 1] = temp;
            }
        }
    }

    printf("\nDisk head movement:\n");

    // Move towards higher cylinders
    for(i = 0; i < n; i++)
    {
        if(req[i] >= head)
        {
            total += abs(req[i] - head);
            printf("%d -> ", head);
            head = req[i];
        }
    }

    // Move to the last cylinder
    total += abs((size - 1) - head);
    printf("%d -> ", size - 1);

    // Jump to the first cylinder
    total += size - 1;
    printf("0 -> ");
    head = 0;

    // Continue servicing remaining requests
    for(i = 0; i < n; i++)
    {
        if(req[i] < head + size && req[i] < req[0])
        {
            total += abs(req[i] - head);
            printf("%d -> ", head);
            head = req[i];
        }
    }

    printf("%d\n", head);

    printf("\nTotal head movement = %d\n", total);

    return 0;
}
